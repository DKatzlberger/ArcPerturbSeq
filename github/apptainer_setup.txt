# Fix GitHub HTTPS auth inside Apptainer + VS Code

# 1. Disable VS Code credential hijacking (IMPORTANT)

export GIT_ASKPASS=
export SSH_ASKPASS=
export VSCODE_GIT_ASKPASS_NODE=
export VSCODE_GIT_ASKPASS_MAIN=
export VSCODE_GIT_ASKPASS_HANDLE=

# 2. Reset Git credential helpers

git config --global --unset-all credential.helper 2>/dev/null
git config --local --unset-all credential.helper 2>/dev/null

# 3. Use simple credential storage

git config --global credential.helper store

# 4. Remove old/expired credentials

rm -f ~/.git-credentials
git credential-cache exit 2>/dev/null

# 5. Trigger login (enter GitHub username + NEW token)

git push
